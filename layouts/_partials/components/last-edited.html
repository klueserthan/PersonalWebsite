{{/* Project override of the kit's "last updated" line. This site builds with enableGitInfo disabled, so .Lastmod falls back to the page's own date unless the front matter sets lastmod explicitly. The line then restated the publication date as an edit date — and at day precision, which on a talk page contradicted the month shown just above it. Show it only when there is a real, different modification date to show. */}} {{- $show := .Params.show_date_updated | default true -}} {{- $is_real := and .Lastmod (ne (.Lastmod.Format "2006-01-02") (.Date.Format "2006-01-02")) -}} {{- if and $show $is_real -}} {{ T "last_updated" }} {{ .Lastmod | time.Format (site.Params.hugoblox.locale.date_format | default ":date_long") -}} {{- else -}}
{{- end -}}
