{{/* Project override of the kit's event-date formatter. Talks on this site are recorded at month precision — every event_start is the first of the month, standing in for "September 2025" — so the kit's day-level format asserted a day that is not in the data, and disagreed with the Talks list, which has always shown the month. A range spanning more than one month still reads as one, e.g. "June – July 2026". */}} {{ $format := "January 2006" }} {{ $start := .Params.event_start | default .Date }} {{ $end := .Params.event_end | default .Params.date_end }} {{ $out := "" }} {{ with $start }} {{ $out = (time .) | time.Format $format }} {{ with $end }} {{ $to := (time .) | time.Format $format }} {{ if ne $to $out }} {{ $out = printf "%s – %s" $out $to }} {{ end }} {{ end }} {{ end }} {{ return $out }}
